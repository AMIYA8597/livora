# Breaking API changes made on purpose, one line each exactly as `python -m app.core.api_contract
# compat` reports it, with a comment above saying why it's safe (for example: no released app
# uses it, or the minimum app version was raised first). CI ignores the lines listed here.

# Role hardening (2026-11-25): staff roles can't be chosen at sign-up and role names are checked.
# Only valid requests are still accepted: there are 4 sign-up roles and every role name is
# under 32 characters, so no real client (web or app) sends anything these limits refuse.
POST /api/v1/admin/users/{user_id}/roles request.role: maxLength tightened from None to 32
POST /api/v1/auth/register request.additional_roles: maxItems tightened from None to 4

# Application checks (2026-11-28): a direct API call stored an application with a rent of -5.
# The web form already caps nothing the API now refuses for real use (an introduction is a few
# sentences; no client sends documents yet), and the web app is the only client in this repo.
POST /api/v1/applications/ request.documents: maxItems tightened from None to 10
POST /api/v1/applications/ request.introduction: maxLength tightened from None to 2000

# Group checks (2026-11-28): a 300-character group name failed with a 500 (the column holds 120).
# The limits match the database columns, so anything they refuse could never have been stored;
# the budget cap is ₹1 crore a month. The web form carries the same limits.
POST /api/v1/groups/ request.description: maxLength tightened from None to 2000
POST /api/v1/groups/ request.name: maxLength tightened from None to 120
POST /api/v1/groups/ request.target_budget_total: maximum tightened from None to 10000000.0
POST /api/v1/groups/ request.target_city: maxLength tightened from None to 100
